$\delta U = \delta Q + \delta W$

$$
\begin{align}
\Delta U  & = \Delta \left( \frac{1}{2} v^{2} \right) + g\Delta h \\
 & = Q + W \\
 U_{2} - U_{1} + \frac{1}{2}v_{2}^{2}v_{1}^{2}+g (h_{2}-h_{1}) & = Q + P_{1}A_{1}l_{1}-P_{2}A_{2}l_{2} \\
  & = Q + P_{1}V_{1} - P_{2}V_{2} = W_{S} \\
  H_{2}-H_{1}+\frac{1}{2}(v_{2}^{2}-v_{2}^{2})+g(l_{2}-l_{1})  & = Q+W_{S}
\end{align}
$$

$$
\begin{align}
U \equiv U(T), \quad PV = hR^{\circ} T \\
H =  U + PV = U + RT \\
H = H(T) \\
\frac{dH}{dT} = \frac{dV}{dT} + R \\
C_{p} = C_{v} + R \rightarrow  \\
1 = \frac{C_{v}}{C_{p}}+\frac{R}{C_{p}} \\
\gamma = \frac{C_{p}}{C_{v}} \\
\frac{R^{\circ}}{C_{p}}= \frac{\gamma-1}{\gamma ^{2}}
\end{align}
$$

$$
\begin{align}
PV = nR^{\circ}T \rightarrow  P=\rho RT
\end{align}
$$
